# Kaggle — test prompt -> 4 views only
#
# Paste this ENTIRE file into ONE new Kaggle "Code" cell and run it.
# It works both on a new Kaggle session (the repository/LGM will be cloned)
# and on a session that already has LGM installed. Do not run a metrics.json
# viewer after this test: prompt-only mode intentionally has no MRC metrics.

import subprocess
import sys
import os
from pathlib import Path

REPO_URL = "https://github.com/huytrao/carve3d-test.git"
BRANCH = "implement_full_pipeline"
WORKDIR = Path("/kaggle/working")
REPO = WORKDIR / "carve3d-test"
LGM_ROOT = WORKDIR / "LGM"
OUTPUT_DIR = WORKDIR / "prompt-smoke-test"

# Keep both Kaggle T4s visible. Prompt-only MVDream is one model and therefore
# runs on GPU 0; GPU 1 is reserved for reconstruction/MRC in the later full
# pipeline run.
os.environ["CUDA_VISIBLE_DEVICES"] = "0,1"

PROMPT = " ".join([
    "a wooden chair",
    "studio product photograph",
    "centered object",
    "white background",
])

# Get the test script. If this is a fresh Kaggle session, clone the branch.
if not REPO.is_dir():
    subprocess.run(
        ["git", "clone", "--quiet", "--depth", "1", "--branch", BRANCH, REPO_URL, str(REPO)],
        check=True,
    )
else:
    subprocess.run(["git", "-C", str(REPO), "fetch", "--quiet", "origin", BRANCH], check=True)
    subprocess.run(["git", "-C", str(REPO), "switch", "--quiet", BRANCH], check=True)
    subprocess.run(
        ["git", "-C", str(REPO), "pull", "--quiet", "--ff-only", "origin", BRANCH],
        check=True,
    )

# A fresh Kaggle session also needs LGM because it provides MVDream's pipeline.
# This downloads the public LGM checkpoint automatically; no checkpoint upload
# is required. Existing LGM installs are reused.
if not (LGM_ROOT / "mvdream" / "pipeline_mvdream.py").is_file():
    subprocess.run(
        ["bash", str(REPO / "scripts" / "setup_lgm_kaggle.sh"), str(LGM_ROOT)],
        check=True,
    )

# Required for LGM/MVDream on Kaggle Python 3.12 (fixes NameError: Union).
subprocess.run(
    [
        sys.executable,
        "-m",
        "pip",
        "install",
        "--quiet",
        "--no-deps",
        "--force-reinstall",
        "kiui==0.2.3",
    ],
    check=True,
)

# Confirm that Kaggle was configured with two T4 GPUs before starting. The
# smoke test still works with one GPU, but tells the user that GPU 1 will not
# be available for the later full pipeline.
gpu_probe = subprocess.run(
    [
        sys.executable,
        "-c",
        "import torch; print('CUDA GPU count:', torch.cuda.device_count()); "
        "[print(f'GPU {i}: {torch.cuda.get_device_name(i)}') "
        "for i in range(torch.cuda.device_count())]",
    ],
    check=True,
)

# Test ONLY: text prompt -> MVDream -> 4 generated views.
# No reconstruction, Gaussian PLY, rendering, MRC, or metrics.json is created.
subprocess.run(
    [
        sys.executable,
        str(REPO / "test_prompt_only.py"),
        "--lgm-root",
        str(LGM_ROOT),
        "--prompt",
        PROMPT,
        "--seed",
        "42",
        "--num-steps",
        "20",
        "--device",
        "0",
        "--output-dir",
        str(OUTPUT_DIR),
    ],
    check=True,
)

print("\nDONE. Open this image in a new Kaggle Code cell:")
print("/kaggle/working/prompt-smoke-test/prompt_grid.png")
