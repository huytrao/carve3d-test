# Paste this entire file into ONE Kaggle Code cell, then run it.
# Kaggle settings: Internet ON, GPU accelerator = T4 x2.
#
# Full executable open workflow:
#   A. 4 real input photos -> LGM -> camera/appearance search -> MRC
#   B. optional research-only prompt -> MVDream -> LGM MRC reward -> LoRA RLFT
#
# This cannot be an exact Carve3D/Instant3D reproduction: their Instant3D
# checkpoint and sparse-view LRM are not released. The repository records this
# honestly in docs/OPEN_RLFT.md and uses public MVDream + LGM instead.

import os
import subprocess
import sys
from pathlib import Path

from IPython.display import display
from PIL import Image, ImageDraw, ImageOps

REPO_URL = "https://github.com/huytrao/carve3d-test.git"
BRANCH = "open_rlft_full_pipeline"
WORKDIR = Path("/kaggle/working")
REPO = WORKDIR / "carve3d-test"
LGM_ROOT = WORKDIR / "LGM"

# Your real, captured four views. Accepted names include view_000.png and
# view_0.png; all four orientations (0, 90, 180, 270) are required.
INPUT_DIR = Path("/kaggle/input/datasets/traoanhuy/carve3d/views")
DIRECT_OUTPUT_DIR = WORKDIR / "carve3d-direct-four-view-calibrated-output"
RLFT_OUTPUT_DIR = WORKDIR / "carve3d-open-rlft-output-v4"

# This is the safe path for YOUR four photographs. It starts from LGM's
# Gaussian prediction, then tests forward-only global colour/opacity/scale
# calibration candidates. It never invokes the unstable CUDA backward kernel.
DIRECT_APPEARANCE_SEARCH = True

# This runs actual LoRA RLFT after direct reconstruction. It trains the public
# MVDream prompt model (GPU 0) with LGM/MRC reward (GPU 1); it is a separate
# paper-style experiment and does not modify the direct four-photo PLY above.
RUN_PROMPT_RLFT = True

# Stabilized quality profile for T4 x2. Each update mixes two prompts with four
# trajectories each and normalizes reward/KL inside each prompt group. Exact
# frozen-base KL, fp32 DDIM log probabilities, and transactional validation
# prevent the near-zero KL and gradient spikes seen in the older v3 run.
RL_EPOCHS = 8
RL_SAMPLES_PER_EPOCH = 8
RL_PROMPTS_PER_UPDATE = 2
RL_DDIM_STEPS = 30  # Official public MVDream/LGM text pipeline setting.
RL_GUIDANCE_SCALE = 7.5
RL_LEARNING_RATE = 1e-5
RL_REWARD_COEFF = 1.0
RL_KL_COEFF = 0.2
RL_TIMESTEP_REDUCTION = "mean"
RL_ADVANTAGE_CLIP = 5.0
RL_MIN_VALIDATION_IMPROVEMENT = 1e-4
RL_MRC_METRIC = "lpips"  # Paper metric. Use "l1" only when LPIPS download is blocked.
RL_EARLY_STOP_PATIENCE = 4
FINAL_CANDIDATES = 8
TRAIN_PROMPTS = [
    "a wooden chair, isolated studio product photograph, white background",
    "a ceramic teapot, isolated studio product photograph, white background",
    "a leather travel suitcase, isolated studio product photograph, white background",
    "a brass desk clock, isolated studio product photograph, white background",
    "a small red toy car, isolated studio product photograph, white background",
    "a blue glass perfume bottle, isolated studio product photograph, white background",
    "a vintage film camera, isolated studio product photograph, white background",
    "a woven picnic basket, isolated studio product photograph, white background",
]
VALIDATION_PROMPTS = [
    "a green desk fan, isolated studio product photograph, white background",
    "a canvas hiking backpack, isolated studio product photograph, white background",
]

# Public model downloads and package setup stay quiet; actual metrics/errors are
# intentionally shown. GPU 0 is MVDream/RL; GPU 1 is LGM/MRC.
os.environ["CUDA_VISIBLE_DEVICES"] = "0,1"
os.environ["HF_HUB_DISABLE_PROGRESS_BARS"] = "1"
os.environ["TQDM_DISABLE"] = "1"
os.environ["TRANSFORMERS_VERBOSITY"] = "error"
os.environ["DIFFUSERS_VERBOSITY"] = "error"

if not REPO.is_dir():
    subprocess.run(
        ["git", "clone", "--quiet", "--depth", "1", "--branch", BRANCH, REPO_URL, str(REPO)],
        check=True,
    )
else:
    subprocess.run(
        [
            "git", "-C", str(REPO), "fetch", "--quiet", "origin",
            f"+refs/heads/{BRANCH}:refs/remotes/origin/{BRANCH}",
        ],
        check=True,
    )
    subprocess.run(
        ["git", "-C", str(REPO), "checkout", "-B", BRANCH, f"origin/{BRANCH}"],
        check=True,
    )

# Preview exactly the four source photos before loading LGM or starting direct
# calibration. This catches reversed/misnamed camera views early.
def resolve_view(angle: int) -> Path:
    aliases = {f"view_{angle:03d}", f"view{angle:03d}", f"view_{angle}", f"view{angle}", str(angle)}
    matches = sorted(
        path for path in INPUT_DIR.iterdir()
        if path.is_file() and path.suffix.lower() in {".png", ".jpg", ".jpeg", ".webp"}
        and path.stem.lower() in aliases
    )
    if len(matches) != 1:
        raise FileNotFoundError(f"Need exactly one image for view_{angle:03d}; found {len(matches)} in {INPUT_DIR}")
    return matches[0]


source_paths = [resolve_view(angle) for angle in (0, 90, 180, 270)]
print("Preview source views before reconstruction/calibration:")
for angle, path in zip((0, 90, 180, 270), source_paths):
    print(f"  {angle:03d}° -> {path}")

preview_size = 320
preview = Image.new("RGB", (preview_size * 2, (preview_size + 28) * 2), "white")
for index, (angle, path) in enumerate(zip((0, 90, 180, 270), source_paths)):
    image = ImageOps.contain(Image.open(path).convert("RGB"), (preview_size, preview_size))
    tile = Image.new("RGB", (preview_size, preview_size), "white")
    tile.paste(image, ((preview_size - image.width) // 2, (preview_size - image.height) // 2))
    x = (index % 2) * preview_size
    y = (index // 2) * (preview_size + 28)
    preview.paste(tile, (x, y))
    ImageDraw.Draw(preview).text((x + 8, y + preview_size + 6), f"view_{angle:03d}", fill="black")
display(preview)

# Stage A. Validates all four real images before downloading/compiling LGM,
# then runs safe direct appearance calibration.
# First run automatically installs LGM quietly and downloads its public weights.
subprocess.run(
    [
        sys.executable,
        str(REPO / "kaggle_import_four_views.py"),
        "--input-dir", str(INPUT_DIR),
        "--output-dir", str(DIRECT_OUTPUT_DIR),
        "--lgm-root", str(LGM_ROOT),
        "--lgm-device", "1",
        "--render-size", "256",
        "--no-orbit",
        "--elevation-candidates", "-10", "-5", "0", "5", "10",
        *( ["--appearance-search"] if DIRECT_APPEARANCE_SEARCH else [] ),
    ],
    check=True,
)

# Stage B + C. Quality RLFT: public MVDream/LGM, reward = -MRC, validation
# best-checkpoint selection, then post-RL output from the best LoRA state.
rlft_command = [
    sys.executable,
    str(REPO / "open_mvdream_rlft.py"),
    "--lgm-root", str(LGM_ROOT),
    "--output-dir", str(RLFT_OUTPUT_DIR),
    "--diffusion-device", "0",
    "--lgm-device", "1",
    "--epochs", str(RL_EPOCHS),
    "--samples-per-epoch", str(RL_SAMPLES_PER_EPOCH),
    "--prompts-per-update", str(RL_PROMPTS_PER_UPDATE),
    "--num-steps", str(RL_DDIM_STEPS),
    "--guidance-scale", str(RL_GUIDANCE_SCALE),
    "--learning-rate", str(RL_LEARNING_RATE),
    "--reward-coeff", str(RL_REWARD_COEFF),
    "--kl-coeff", str(RL_KL_COEFF),
    "--timestep-loss-reduction", RL_TIMESTEP_REDUCTION,
    "--advantage-clip", str(RL_ADVANTAGE_CLIP),
    "--min-validation-improvement", str(RL_MIN_VALIDATION_IMPROVEMENT),
    "--mrc-metric", RL_MRC_METRIC,
    "--early-stop-patience", str(RL_EARLY_STOP_PATIENCE),
    "--transactional-validation",
    "--final-candidates", str(FINAL_CANDIDATES),
    "--final-prompt", "a wooden chair, isolated studio product photograph, white background",
]
for prompt in TRAIN_PROMPTS:
    rlft_command.extend(["--prompt", prompt])
for prompt in VALIDATION_PROMPTS:
    rlft_command.extend(["--validation-prompt", prompt])
if RUN_PROMPT_RLFT:
    subprocess.run(rlft_command, check=True)
else:
    print("Skipped prompt RLFT: it cannot improve reconstruction from your four real photos.")

print("\nCOMPLETED")
print("Direct four-image reconstruction:", DIRECT_OUTPUT_DIR)
if RUN_PROMPT_RLFT:
    print("Open RLFT output/checkpoint:", RLFT_OUTPUT_DIR)
    print("Post-RL MRC:", RLFT_OUTPUT_DIR / "rlft_metrics.json")
