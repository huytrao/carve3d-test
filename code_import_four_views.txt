import subprocess
import sys
from pathlib import Path

REPO = Path("/kaggle/working/carve3d-test")
REPO_URL = "https://github.com/huytrao/carve3d-test.git"
BRANCH = "import_direct_four_views"
INPUT_DIR = Path("/kaggle/input/datasets/traoanhuy/carve3d/views")
OUTPUT_DIR = Path("/kaggle/working/carve3d-four-view-output")
LGM_ROOT = Path("/kaggle/working/LGM")

if not REPO.is_dir():
    subprocess.run(
        ["git", "clone", "--quiet", "--depth", "1", "--branch", BRANCH, REPO_URL, str(REPO)],
        check=True,
    )
else:
    # A previous Kaggle run may have cloned only another branch. Fetch the
    # remote-tracking ref explicitly, then create/reset this local branch.
    # `checkout -B` is intentional: unlike `git switch branch`, it also works
    # when this repository has never had a local import_direct_four_views ref.
    subprocess.run(
        [
            "git", "-C", str(REPO), "fetch", "--quiet", "origin",
            f"+refs/heads/{BRANCH}:refs/remotes/origin/{BRANCH}",
        ],
        check=True,
    )
    subprocess.run(
        ["git", "-C", str(REPO), "checkout", "-B", BRANCH, f"origin/{BRANCH}"],
        check=True,
    )

if str(REPO) not in sys.path:
    sys.path.insert(0, str(REPO))

# This validates view_000, view_090, view_180, view_270 BEFORE installing LGM.
# On two T4 GPUs LGM is automatically placed on GPU 1.
from kaggle_import_four_views import main as run_four_real_views

run_four_real_views(
    [
        "--input-dir", str(INPUT_DIR),
        "--output-dir", str(OUTPUT_DIR),
        "--lgm-root", str(LGM_ROOT),
    ]
)
